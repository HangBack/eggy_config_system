# 使用示例

## 角色表

### schema/Character.json

```json
{
  "id": {"label": "ID", "type": "number", "primary": true},
  "name": {"label": "名称", "type": "text"},
  "class": {"label": "职业", "type": "option", "enum": "CharacterClass"},
  "rarity": {"label": "稀有度", "type": "option", "enum": "ItemRarity"},
  "baseStats": {
    "label": "基础属性",
    "type": "entry",
    "fields": {
      "hp": {"label": "生命", "type": "number", "default": 100},
      "atk": {"label": "攻击", "type": "number", "default": 10},
      "def": {"label": "防御", "type": "number", "default": 5}
    }
  },
  "skills": {"label": "技能列表", "type": "list", "itemType": "number"},
  "description": {"label": "描述", "type": "richtext"}
}
```

### enum/CharacterClass.json

```json
{
  "name": "CharacterClass",
  "values": {
    "Warrior": 1,
    "Mage": 2,
    "Archer": 3,
    "Healer": 4
  }
}
```

## 技能表

### schema/Skill.json

```json
{
  "id": {"label": "ID", "type": "number", "primary": true},
  "name": {"label": "名称", "type": "richtext"},
  "category": {"label": "分类", "type": "option", "enum": "SkillCategory"},
  "damage": {"label": "伤害", "type": "number"},
  "cooldown": {"label": "冷却(秒)", "type": "number", "default": 1},
  "effects": {"label": "效果", "type": "flags", "enum": "SkillEffect"}
}
```

### enum/SkillEffect.json

```json
{
  "name": "SkillEffect",
  "isFlags": true,
  "values": {
    "Burn": 0,
    "Freeze": 1,
    "Stun": 2,
    "Poison": 3
  }
}
```

---

## 物品表

### schema/Item.json

```json
{
  "id": {"label": "ID", "type": "number", "primary": true},
  "name": {"label": "名称", "type": "text"},
  "type": {"label": "类型", "type": "option", "enum": "ItemType"},
  "rarity": {"label": "稀有度", "type": "option", "enum": "ItemRarity"},
  "price": {"label": "价格", "type": "number", "default": 0},
  "stackable": {"label": "可堆叠", "type": "boolean", "default": true},
  "iconColor": {"label": "图标颜色", "type": "color", "default": "#FFFFFF"},
  "linkedSkill": {
    "label": "关联技能",
    "type": "datalist",
    "source": "Skill",
    "valueField": "id",
    "labelField": "name"
  }
}
```
